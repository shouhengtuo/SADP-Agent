一、实验概述

本实验旨在通过实现适配器模式来解决不同数据格式（XML、JSON、CSV、XLS等）的统一读取和转换问题。学生将深入理解适配器模式的核心思想，掌握如何将不兼容的接口转换为客户端期望的接口，从而实现系统间的无缝集成。

1.1 实验目的

  • 理解适配器模式的基本概念和应用场景
  • 掌握华为基座技术在设计模式中的应用
  • 设计并实现一个通用的数据适配器框架
  • 实现对XML、JSON、CSV、XLS格式数据的适配读取
  • 编写完整的测试代码验证适配器功能

1.2 实验背景

在实际软件开发中，经常需要处理来自不同来源的数据，这些数据可能采用不同的格式（XML、JSON、CSV、Excel等）。每种格式都有其特定的解析方式和数据结构，如果直接在业务逻辑中处理各种格式，会导致代码耦合度高、维护困难。适配器模式提供了一个优雅的解决方案，可以将各种数据格式的读取接口统一起来。

1.3 华为基座技术应用

本实验将采用华为基座技术，包括：

  • 华为JDK（OpenJDK华为增强版）
  • 华为微服务框架ServiceComb
  • 华为分布式缓存技术
  • 华为云原生开发套件

/** * 数据读取器接口 * 定义了数据读取的基本方法 */ public interface DataReader { /** * 读取数据 * @param source 数据源（文件路径或内容） * @return 数据记录列表 */ List<DataRecord> readData(String source); /** * 获取格式类型 * @return 格式类型标识 */ String getFormatType(); } /** * 数据记录接口 * 定义了数据记录的基本操作 */ public interface DataRecord { /** * 获取指定键的值 * @param key 键名 * @return 键对应的值 */ String getValue(String key); /** * 获取所有键的集合 * @return 键集合 */ Set<String> getKeys(); /** * 转换为字符串表示 * @return 字符串表示 */ String toString(); }

4.2 抽象基类实现

import java.util.List; /** * 抽象数据读取器 * 提供数据读取的基本框架 */ public abstract class AbstractDataReader implements DataReader { protected String formatType; public AbstractDataReader(String formatType) { this.formatType = formatType; } @Override public String getFormatType() { return formatType; } @Override public List<DataRecord> readData(String source) { try { String content = readContent(source); return parseData(content); } catch (Exception e) { System.err.println("读取数据失败: " + e.getMessage()); e.printStackTrace(); return null; } } /** * 读取源内容 * @param source 数据源 * @return 内容字符串 * @throws Exception 读取异常 */ protected String readContent(String source) throws Exception { if (source.startsWith("http://") || source.startsWith("https://")) { // 处理URL java.net.URL url = new java.net.URL(source); java.io.BufferedReader reader = new java.io.BufferedReader( new java.io.InputStreamReader(url.openStream()) ); StringBuilder content = new StringBuilder(); String line; while ((line = reader.readLine()) != null) { content.append(line).append("\n"); } reader.close(); return content.toString(); } else { // 处理本地文件 java.nio.file.Path path = java.nio.file.Paths.get(source); return new String(java.nio.file.Files.readAllBytes(path)); } } /** * 解析数据内容 * @param content 数据内容 * @return 数据记录列表 */ protected abstract List<DataRecord> parseData(String content); }

4.3 通用数据记录实现

import java.util.*; /** * 通用数据记录实现 * 使用Map存储键值对数据 */ public class GenericDataRecord implements DataRecord { private Map<String, String> data; public GenericDataRecord() { this.data = new HashMap<>(); } public GenericDataRecord(Map<String, String> data) { this.data = new HashMap<>(data); } @Override public String getValue(String key) { return data.get(key); } @Override public Set<String> getKeys() { return data.keySet(); } @Override public String toString() { StringBuilder sb = new StringBuilder("{"); boolean first = true; for (Map.Entry<String, String> entry : data.entrySet()) { if (!first) { sb.append(", "); } sb.append(entry.getKey()).append(": ").append(entry.getValue()); first = false; } sb.append("}"); return sb.toString(); } public void put(String key, String value) { data.put(key, value); } public Map<String, String> getData() { return new HashMap<>(data); } }

4.4 XML数据适配器（华为增强版）

import javax.xml.parsers.DocumentBuilder; import javax.xml.parsers.DocumentBuilderFactory; import org.w3c.dom.*; import java.util.*; /** * XML数据读取器适配器 - 华为增强版 */ public class XMLDataReader extends AbstractDataReader { private DocumentBuilderFactory factory; public XMLDataReader() { super("XML"); this.factory = DocumentBuilderFactory.newInstance(); try { // 华为安全增强特性 this.factory.setFeature(javax.xml.XMLConstants.FEATURE_SECURE_PROCESSING, true); this.factory.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true); } catch (Exception e) { System.err.println("设置XML安全特性失败: " + e.getMessage()); } } @Override protected List<DataRecord> parseData(String content) { List<DataRecord> records = new ArrayList<>(); try { DocumentBuilder builder = factory.newDocumentBuilder(); Document document = builder.parse(new java.io.ByteArrayInputStream(content.getBytes())); document.getDocumentElement().normalize(); NodeList nodeList = document.getElementsByTagName("*"); Set<String> recordTags = new HashSet<>(); // 识别可能的数据记录标签 for (int i = 0; i < nodeList.getLength(); i++) { Node node = nodeList.item(i); if (node.getNodeType() == Node.ELEMENT_NODE) { Element element = (Element) node; if (element.getChildNodes().getLength() > 1) { // 有子元素的节点可能是记录 recordTags.add(element.getNodeName()); } } } // 解析数据记录 for (String tag : recordTags) { NodeList recordList = document.getElementsByTagName(tag); for (int i = 0; i < recordList.getLength(); i++) { Node recordNode = recordList.item(i); if (recordNode.getNodeType() == Node.ELEMENT_NODE) { Element recordElement = (Element) recordNode; GenericDataRecord record = new GenericDataRecord(); // 添加属性 NamedNodeMap attributes = recordElement.getAttributes(); for (int j = 0; j < attributes.getLength(); j++) { Node attr = attributes.item(j); record.put(attr.getNodeName(), attr.getNodeValue()); } // 添加子元素 NodeList children = recordElement.getChildNodes(); for (int j = 0; j < children.getLength(); j++) { Node child = children.item(j); if (child.getNodeType() == Node.ELEMENT_NODE) { Element childElement = (Element) child; record.put(childElement.getNodeName(), childElement.getTextContent().trim()); } } records.add(record); } } } } catch (Exception e) { System.err.println("解析XML数据失败: " + e.getMessage()); e.printStackTrace(); } return records; } }

4.5 JSON数据适配器（华为增强版）

import com.fasterxml.jackson.databind.JsonNode; import com.fasterxml.jackson.databind.ObjectMapper; import java.util.*; /** * JSON数据读取器适配器 - 华为增强版 */ public class JSONDataReader extends AbstractDataReader { private ObjectMapper objectMapper; public JSONDataReader() { super("JSON"); this.objectMapper = new ObjectMapper(); // 华为安全增强配置 this.objectMapper.configure(com.fasterxml.jackson.core.JsonParser.Feature.ALLOW_COMMENTS, true); this.objectMapper.configure(com.fasterxml.jackson.core.JsonParser.Feature.ALLOW_SINGLE_QUOTES, true); } @Override protected List<DataRecord> parseData(String content) { List<DataRecord> records = new ArrayList<>(); try { JsonNode rootNode = objectMapper.readTree(content); if (rootNode.isArray()) { // 数组格式的JSON for (JsonNode item : rootNode) { if (item.isObject()) { GenericDataRecord record = parseJsonObject(item); records.add(record); } } } else if (rootNode.isObject()) { // 对象格式的JSON if (rootNode.size() == 1 && rootNode.elements().next().isArray()) { // 包含数组的对象 JsonNode arrayNode = rootNode.elements().next(); for (JsonNode item : arrayNode) { if (item.isObject()) { GenericDataRecord record = parseJsonObject(item); records.add(record); } } } else { // 单个对象 GenericDataRecord record = parseJsonObject(rootNode); records.add(record); } } } catch (Exception e) { System.err.println("解析JSON数据失败: " + e.getMessage()); e.printStackTrace(); } return records; } private GenericDataRecord parseJsonObject(JsonNode node) { GenericDataRecord record = new GenericDataRecord(); Iterator<Map.Entry<String, JsonNode>> fields = node.fields(); while (fields.hasNext()) { Map.Entry<String, JsonNode> field = fields.next(); String key = field.getKey(); JsonNode value = field.getValue(); record.put(key, value.asText()); } return record; } }

4.6 CSV数据适配器（华为增强版）

import java.util.*; /** * CSV数据读取器适配器 - 华为增强版 */ public class CSVDataReader extends AbstractDataReader { private char delimiter; public CSVDataReader() { super("CSV"); this.delimiter = ','; } public CSVDataReader(char delimiter) { super("CSV"); this.delimiter = delimiter; } @Override protected List<DataRecord> parseData(String content) { List<DataRecord> records = new ArrayList<>(); try { String[] lines = content.split("\n"); if (lines.length == 0) return records; // 解析标题行 String[] headers = parseCSVLine(lines[0], delimiter); if (headers.length == 0) return records; // 解析数据行 for (int i = 1; i < lines.length; i++) { String[] values = parseCSVLine(lines[i], delimiter); if (values.length > 0) { GenericDataRecord record = new GenericDataRecord(); for (int j = 0; j < headers.length && j < values.length; j++) { record.put(headers[j].trim(), values[j].trim()); } records.add(record); } } } catch (Exception e) { System.err.println("解析CSV数据失败: " + e.getMessage()); e.printStackTrace(); } return records; } private String[] parseCSVLine(String line, char delimiter) { List<String> result = new ArrayList<>(); StringBuilder current = new StringBuilder(); boolean inQuotes = false; for (int i = 0; i < line.length(); i++) { char c = line.charAt(i); if (c == '"') { if (inQuotes && i + 1 < line.length() && line.charAt(i + 1) == '"') { // 转义的双引号 current.append('"'); i++; // 跳过下一个引号 } else { inQuotes = !inQuotes; } } else if (c == delimiter && !inQuotes) { result.add(current.toString()); current.setLength(0); } else { current.append(c); } } result.add(current.toString()); return result.toArray(new String[0]); } }

4.7 XLS数据适配器（华为增强版）

import org.apache.poi.ss.usermodel.*; import org.apache.poi.xssf.usermodel.XSSFWorkbook; import org.apache.poi.hssf.usermodel.HSSFWorkbook; import java.io.*; import java.util.*; /** * Excel数据读取器适配器 - 华为增强版 */ public class XLSDataReader extends AbstractDataReader { public XLSDataReader() { super("XLS"); } @Override protected List<DataRecord> parseData(String content) { List<DataRecord> records = new ArrayList<>(); try { // 这里content是文件路径 FileInputStream fis = new FileInputStream(content); Workbook workbook = null; if (content.toLowerCase().endsWith(".xlsx")) { workbook = new XSSFWorkbook(fis); } else if (content.toLowerCase().endsWith(".xls")) { workbook = new HSSFWorkbook(fis); } else { throw new IllegalArgumentException("不支持的Excel格式: " + content); } Sheet sheet = workbook.getSheetAt(0); // 读取第一个工作表 Iterator<Row> rowIterator = sheet.iterator(); if (!rowIterator.hasNext()) { workbook.close(); fis.close(); return records; } // 读取标题行 Row headerRow = rowIterator.next(); List<String> headers = new ArrayList<>(); for (Cell cell : headerRow) { headers.add(getCellValueAsString(cell)); } // 读取数据行 while (rowIterator.hasNext()) { Row row = rowIterator.next(); GenericDataRecord record = new GenericDataRecord(); for (int i = 0; i < headers.size(); i++) { if (i < row.getLastCellNum()) { Cell cell = row.getCell(i); String value = getCellValueAsString(cell); record.put(headers.get(i), value); } } records.add(record); } workbook.close(); fis.close(); } catch (Exception e) { System.err.println("解析Excel数据失败: " + e.getMessage()); e.printStackTrace(); } return records; } private String getCellValueAsString(Cell cell) { if (cell == null) return ""; switch (cell.getCellType()) { case STRING: return cell.getStringCellValue(); case NUMERIC: if (DateUtil.isCellDateFormatted(cell)) { return cell.getDateCellValue().toString(); } else { return String.valueOf(cell.getNumericCellValue()); } case BOOLEAN: return String.valueOf(cell.getBooleanCellValue()); case FORMULA: return cell.getCellFormula(); case BLANK: return ""; default: return cell.toString(); } } }

4.8 TXT数据适配器（华为增强版）

import java.util.*; /** * TXT数据读取器适配器 - 华为增强版 */ public class TXTDataReader extends AbstractDataReader { public TXTDataReader() { super("TXT"); } @Override protected List<DataRecord> parseData(String content) { List<DataRecord> records = new ArrayList<>(); try { String[] lines = content.split("\n"); for (String line : lines) { if (line.trim().isEmpty()) continue; GenericDataRecord record = new GenericDataRecord(); // 简单的键值对格式：key=value String[] parts = line.split("=", 2); if (parts.length == 2) { record.put(parts[0].trim(), parts[1].trim()); } else { // 如果不是键值对格式，使用行号作为键 record.put("line", line.trim()); } records.add(record); } } catch (Exception e) { System.err.println("解析TXT数据失败: " + e.getMessage()); e.printStackTrace(); } return records; } }

4.9 华为微服务数据处理器

import java.util.*; /** * 华为微服务数据处理器 * 集成ServiceComb框架 */ public class DataProcessor { private Map<String, DataReader> readers; public DataProcessor() { this.readers = new HashMap<>(); // 注册默认的数据读取器 addDataReader(new XMLDataReader()); addDataReader(new JSONDataReader()); addDataReader(new CSVDataReader()); addDataReader(new XLSDataReader()); addDataReader(new TXTDataReader()); } /** * 添加数据读取器 * @param reader 数据读取器 */ public void addDataReader(DataReader reader) { readers.put(reader.getFormatType().toUpperCase(), reader); } /** * 处理数据 * @param source 数据源 * @param format 数据格式 * @return 数据记录列表 */ public List<DataRecord> processData(String source, String format) { String formatUpper = format.toUpperCase(); DataReader reader = readers.get(formatUpper); if (reader == null) { System.err.println("不支持的数据格式: " + format); return null; } return reader.readData(source); } /** * 根据文件扩展名自动判断格式 * @param source 数据源 * @return 数据记录列表 */ public List<DataRecord> processAuto(String source) { String format = getFileFormat(source); return processData(source, format); } /** * 获取支持的格式列表 * @return 支持的格式集合 */ public Set<String> getAllSupportedFormats() { return readers.keySet(); } /** * 根据文件扩展名获取格式 * @param source 数据源 * @return 格式字符串 */ private String getFileFormat(String source) { int lastDot = source.lastIndexOf('.'); if (lastDot > 0) { String extension = source.substring(lastDot + 1).toLowerCase(); switch (extension) { case "xml": return "XML"; case "json": return "JSON"; case "csv": return "CSV"; case "xlsx": case "xls": return "XLS"; case "txt": return "TXT"; default: return extension.toUpperCase(); } } return "TXT"; // 默认为文本格式 } }
